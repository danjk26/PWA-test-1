// Check GitHub server for new sw.js updates (bypassing browser HTTP cache) function checkForAppUpdate() { const updateBtn = document.getElementById('update-btn'); updateBtn.innerText = 'Checking GitHub...'; if ('serviceWorker' in navigator) { navigator.serviceWorker.getRegistration().then(registration => { if (registration) { // Force registration update with cache-busting timestamp fetch('./sw.js?t=' + Date.now(), { cache: 'no-store' }) .then(() => registration.update()) .then(() => { if (registration.waiting) { registration.waiting.postMessage('skipWaiting'); alert('New update found! Reloading app...'); window.location.reload(); } else { registration.addEventListener('updatefound', () => { const newWorker = registration.installing; newWorker.addEventListener('statechange', () => { if (newWorker.state === 'installed') { alert('New update installed! Reloading app...'); window.location.reload(); } }); }); setTimeout(() => { updateBtn.innerText = 'Check for App Update'; alert('App is already up to date!'); }, 1200); } }) .catch(err => { updateBtn.innerText = 'Check for App Update'; alert('Could not check for updates. Make sure you are online.'); }); } else { updateBtn.innerText = 'Check for App Update'; alert('No Service Worker active.'); } }); } else { alert('Service Worker is not supported in this browser.'); } }
